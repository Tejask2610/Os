FILE="addressbook.txt" 
 
createAddressBook() 
{ 
  if [ -e "$FILE"] ; 
  then  
       echo "Address book already exists." 
  else 
       touch "$FILE" 
       echo "Address book created." 
  fi         
}

{ 
  if [! -s "$FILE"];

}
