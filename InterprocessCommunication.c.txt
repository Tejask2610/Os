#include <stdio.h> 
#include <stdlib.h> 
#include <string.h> 
#include <unistd.h> 
#include <sys/types.h> 
#include <sys/stat.h> 
#include <fcntl.h> 
#define FIFO1 "/tmp/fifo1" 
#define FIFO2 "/tmp/fifo2" 
#define BUFFER_SIZE 1024 
void process1() { 
int fd_write, fd_read; 
char input_string[BUFFER_SIZE]; 
char output_string[BUFFER_SIZE]; 
printf("[Process 1] Enter sentences (Press Enter then Ctrl+D to finish):\n"); 
// Read input from user until EOF 
char temp[BUFFER_SIZE]; 
input_string[0] = '\0'; 
while (fgets(temp, BUFFER_SIZE, stdin) != NULL) { 
strcat(input_string, temp); 
} 
// Open FIFO1 for writing user input 
fd_write = open(FIFO1, O_WRONLY); 
write(fd_write, input_string, strlen(input_string) + 1); 
close(fd_write); 
// Open FIFO2 for reading the processed results 
fd_read = open(FIFO2, O_RDONLY); 
read(fd_read, output_string, BUFFER_SIZE); 
close(fd_read); 
// Display the received content on standard output 
printf("\n[Process 1] Received Results from Process 2:\n%s", output_string); 
} 
void process2() { 
    int fd_read, fd_write; 
    char buffer[BUFFER_SIZE]; 
    int characters = 0, words = 0, lines = 0; 
    FILE *file; 
 
    // Open FIFO1 to read the sentence from Process 1 
    fd_read = open(FIFO1, O_RDONLY); 
    read(fd_read, buffer, BUFFER_SIZE); 
    close(fd_read); 
 
    // Count characters, words, and lines 
    int i = 0; 
    int in_word = 0; 
    while (buffer[i] != '\0') { 
        characters++; 
 
        if (buffer[i] == '\n') { 
            lines++; 
        } 
 
        if (buffer[i] == ' ' || buffer[i] == '\t' || buffer[i] == '\n') { 
            in_word = 0; 
        } else if (in_word == 0) { 
            in_word = 1; 
            words++; 
        } 
        i++; 
    } 
    // If the text doesn't end with a newline but has content, count it as a line 
    if (characters > 0 && buffer[i-1] != '\n') { 
        lines++; 
    } 
 
    // Write output to a text file 
    file = fopen("output.txt", "w"); 
    if (file == NULL) { 
        perror("Error opening file"); 
        exit(1); 
    } 
    fprintf(file, "Character count: %d\n", characters); 
    fprintf(file, "Word count: %d\n", words); 
    fprintf(file, "Line count: %d\n", lines); 
    fclose(file); 
 
    // Read the contents back from the text file 
    char file_content[BUFFER_SIZE]; 
    char line[128]; 
    file_content[0] = '\0'; 
 
    file = fopen("output.txt", "r"); 
while (fgets(line, sizeof(line), file) != NULL) { 
strcat(file_content, line); 
} 
fclose(file); 
// Open FIFO2 to send file contents back to Process 1 
fd_write = open(FIFO2, O_WRONLY); 
write(fd_write, file_content, strlen(file_content) + 1); 
close(fd_write); 
} 
int main() { 
// Create the named pipes (FIFOs) 
mkfifo(FIFO1, 0666); 
mkfifo(FIFO2, 0666); 
pid_t pid = fork(); 
if (pid < 0) { 
perror("Fork failed"); 
return 1; 
} 
if (pid > 0) { 
// Parent execution block 
process1(); 
} else { 
// Child execution block 
process2(); 
exit(0); 
} 
// Clean up FIFOs after communication ends 
unlink(FIFO1); 
unlink(FIFO2); 
return 0; 
}
